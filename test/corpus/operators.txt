===
Delphi operator implementation with TypeName.OperatorName
===

class operator TVector.Implicit(const Value: Double): TVector;
begin
  Result.X := Value;
end;

class operator TVector.Add(const A, B: TVector): TVector;
begin
  Result.X := A.X + B.X;
end;

    class operator Inc(const V: TMyRecord): TMyRecord;
    class operator Dec(const V: TMyRecord): TMyRecord;
