================================================================================
Error Recovery Tests
================================================================================

These tests verify that tree-sitter produces specific MISSING nodes for common
syntax errors, rather than generic ERROR nodes.

================================================================================

:skip
================================================================================

var
  a: Integer
  b: String;

---

(root
  (codeFragment
    (declVars
      (kVar)
      (declVar
        (identifier)
        (type
          (typeref
            (identifier)))
        (MISSING ";"))
      (declVar
        (identifier)
        (type
          (typeref
            (identifier)))))))
